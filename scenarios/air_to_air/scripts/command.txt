// Basic Functions
script WsfPlatform FindValidPlatform(string aPlatformName)
    WsfPlatform plat = WsfSimulation.FindPlatform(aPlatformName);
    if(plat.IsNull() || !plat.IsValid())
    {
        print(aPlatformName, " is not exist");
    }
    return plat;
end_script

script WsfWeapon FindReadyWeapon(WsfPlatform aPlatform, string aWeaponName)
    WsfWeapon weapon = aPlatform.Weapon(aWeaponName);
    if(weapon.IsNull())
    {
        print("weapon is not exist");
        return weapon;
    }
    if(weapon.QuantityRemaining() == 0)
    {
        print("weapon quantity is 0");
    }
    return weapon;
end_script

script void AttackTarget(string aPlatformName,       # Platform name (e.g., "blue_2")
                         string aWeaponName,
                         string aTargetName)
    WsfPlatform plat = FindValidPlatform(aPlatformName);
    WsfPlatform target = FindValidPlatform(aTargetName);
    if(plat.IsNull() || !plat.IsValid() || target.IsNull() || !target.IsValid())
    {
      print("plat or target is not exist");
      return;
    }
    WsfWeapon weapon = FindReadyWeapon(plat, aWeaponName);
    if(weapon.IsNull() || weapon.QuantityRemaining() == 0)
    {
       return;
    }
    foreach (WsfTrack atrack in plat.MasterTrackList())
    {
       if (atrack.IsValid() && !atrack.IsNull() && atrack.TargetName() == aTargetName)
       {
          weapon.CueToTarget(atrack);
          weapon.Fire(atrack);
          print("weapon fired");
          return;
       }
    }
    print("target track not found");
end_script

script void  CoAttackTarget(string aPlatformName,
                            string aWeaponName,
                            string aTargetName,
                            string aGuiderName)
    WsfPlatform plat = FindValidPlatform(aPlatformName);
    WsfPlatform target = FindValidPlatform(aTargetName);
    WsfPlatform guider = FindValidPlatform(aGuiderName);
    if(plat.IsNull() || !plat.IsValid() ||
       target.IsNull() || !target.IsValid() ||
       guider.IsNull() || !guider.IsValid())
    {
        print("plat or target or guider is not exist");
        return;
    }
    WsfWeapon weapon = FindReadyWeapon(plat, aWeaponName);
    if(weapon.IsNull() || weapon.QuantityRemaining() == 0)
    {
        return;
    }
    foreach(WsfTrack track in guider.MasterTrackList())
    {
        if(track.TargetName() == aTargetName)
        {
            weapon.CueToTarget(track);
            weapon.Fire(track);
            print("weapon fired");
            return;
        }
    }
    print("target track not found");
end_script

script void FlyInPath(string aPlatformName,      
                      string aRoute,
                      bool bIsCycle)
                      
   WsfPlatform plat = FindValidPlatform(aPlatformName);
   if(!plat.IsValid())
   {
      return;
   }
   Array<string> point_array = aRoute.Split(";");
   WsfRoute path = WsfRoute();
   int size = point_array.Size();
   for(int iter = 0;iter<size;iter+=1)
   {
      Array<string> point = point_array[iter].Split(","); 
      WsfWaypoint way_point = WsfWaypoint.Create((double)point[0],
                                                 (double)point[1],
                                                 (double)point[2],
                                                 (double)point[3],
                                                 "LATITUDE_AND_LONGITUDE");
      if(iter == 0 && bIsCycle)
      {
         way_point.SetLabel("start");
      }
      if(iter == size -1 &&bIsCycle)
      {
         way_point.SetGoToLabel("start");
      }
      path.Append(way_point);
   }
   plat.FollowRoute(path);   
end_script

script void LauncherSensorCue(string aPlatformName,
                              double heading)
   WsfPlatform plat = FindValidPlatform(aPlatformName);
   if(plat.IsNull() || !plat.IsValid() ||
      (!plat.CategoryMemberOf("ground_launcher") 
    && !plat.CategoryMemberOf("ground_radar")
    && !plat.CategoryMemberOf("destroyer")))
   {
      return;
   }
   WsfSensor sensor = plat.SensorEntry(0);
   if(!sensor.IsValid())
   {
      return;
   }
   sensor.CueToAbsoluteAzEl(heading,0);

end_script

script bool Takeoff(string aPlatformName, string aTakeOffPlatformName,
                    string aTakeOffPlatformType, double fLatitude,
                    double fLongitude, double fAltitude, double fSpeed)
    WsfPlatform plat = WsfSimulation.FindPlatform(aPlatformName);
    if(plat.IsNull() || !plat.IsValid() ||
      (!plat.CategoryMemberOf("aircraft_carrier") &&
       !plat.CategoryMemberOf("airport")))
    {
        return false;
    }
    Array<Object> args = Array<Object>();
    args.PushBack(aTakeOffPlatformName);
    args.PushBack(aTakeOffPlatformType);
    args.PushBack(fLatitude);
    args.PushBack(fLongitude);
    args.PushBack(fAltitude);
    args.PushBack(fSpeed);
    plat.ExecuteScriptWithArgs("TakeOff", args);
    return true;
end_script

script void CreatePlatform(string id,string type,string side,
                       double heading,double pitch,double roll,
                       double lat,double lon,double alt,double speed)
   WsfPlatform plat = WsfSimulation.CreatePlatform(type);
   if(!plat.IsValid())
   {
      return;
   }

   plat.SetSide(side);
   WsfPlatform point = WsfSimulation.AddPlatform(plat,id);
   point.SetLocation(lat,lon,alt);
   point.GoToSpeed(speed);
   point.TurnToHeading(heading);
   point.GoToAltitude(alt);
end_script

script void PumpManeuver(
    string aPlatformName,       # Platform name (e.g., "blue_2")
    double fAltitude,           # Flight altitude (meters)
    double fHeading,            # Heading (degrees)
    double fPitchAngle,
    double fMach,               # Mach number
    double fOverload,           # G-load
    bool bBanAfterburner,       # Ban afterburner flag
    double fAfterburnerLimit,   # Afterburner limit percentage
    int nFrameAngleTargets,     # Frame angle constraint target count
    string aTargetName,
    string aFireMatePlatformName
)
    WsfPlatform plat = WsfSimulation.FindPlatform(aPlatformName);
    WsfSixDOF_Mover mover = (WsfSixDOF_Mover) plat.Mover();
    mover.SetPitchGLoadMax(fOverload);
    plat.GoToAltitude(fAltitude);
    plat.TurnToHeading(fHeading);
    plat.GoToMachNumber(fMach);
    print(aPlatformName, " PumpAltHdgControl executed-----------");
end_script
